-- 1. Definición de Tipos Enumerados (Enums)
CREATE TYPE tipo_rol AS ENUM ('ADMINISTRATIVO', 'ADMINISTRADOR');
CREATE TYPE clase_licencia AS ENUM ('A', 'B', 'C', 'D', 'E', 'F', 'G');
CREATE TYPE tipo_tramite AS ENUM ('EMISION', 'RENOVACION', 'DUPLICADO', 'TRIPLICADO');
CREATE TYPE estado_licencia AS ENUM ('VIGENTE', 'EXPIRADA', 'NO_VIGENTE_CAMBIO_DATOS');

-- 2. Tabla de Usuarios (Operadores del Sistema)
CREATE TABLE usuarios (
    id BIGSERIAL PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL, -- Para hashing con BCrypt
    apellido VARCHAR(100) NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    rol tipo_rol NOT NULL DEFAULT 'ADMINISTRATIVO',
    activo BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. Tabla de Titulares 
CREATE TABLE titulares (
    id BIGSERIAL PRIMARY KEY,
    tipo_documento VARCHAR(10) NOT NULL, -- DNI, LE, LC, PASAPORTE
    numero_documento VARCHAR(20) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    fecha_nacimiento DATE NOT NULL,
    direccion VARCHAR(255) NOT NULL,
    grupo_sanguineo VARCHAR(3) NOT NULL, -- A, B, AB, O
    factor_rh VARCHAR(2) NOT NULL,       -- +, -
    donante_organos BOOLEAN NOT NULL,
    created_at TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    -- Restricción de unicidad para el documento de identidad
    CONSTRAINT uq_titular_documento UNIQUE (tipo_documento, numero_documento)
);

-- 4. Tabla de Licencias 
CREATE TABLE licencias (
    id BIGSERIAL PRIMARY KEY,
    titular_id BIGINT NOT NULL,
    clase clase_licencia NOT NULL,
    fecha_inicio DATE NOT NULL, -- Fecha del sistema
    fecha_vencimiento DATE NOT NULL,
    observaciones TEXT, -- Limitaciones o restricciones del conductor
    estado estado_licencia NOT NULL DEFAULT 'VIGENTE',
    created_at TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT fk_licencias_titular FOREIGN KEY (titular_id) 
        REFERENCES titulares (id) ON DELETE RESTRICT
);

-- 5. Tabla de Trámites 
CREATE TABLE tramites (
    id BIGSERIAL PRIMARY KEY,
    licencia_id BIGINT NOT NULL,
    usuario_id BIGINT NOT NULL, -- Administrativo responsable
    tipo_tramite tipo_tramite NOT NULL,
    fecha_tramite TIMESTAMP WITHOUT TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    costo_base NUMERIC(10, 2) NOT NULL,
    gastos_administrativos NUMERIC(10, 2) NOT NULL, -- Se obtiene de la configuración vigente al momento del trámite
    costo_total NUMERIC(10, 2) GENERATED ALWAYS AS (costo_base + gastos_administrativos) STORED,
    
    CONSTRAINT fk_tramites_licencia FOREIGN KEY (licencia_id) 
        REFERENCES licencias (id) ON DELETE RESTRICT,
    CONSTRAINT fk_tramites_usuario FOREIGN KEY (usuario_id) 
        REFERENCES usuarios (id) ON DELETE RESTRICT
);

-- 6. Tabla Paramétrica de Tarifas (Matriz de Costos)
CREATE TABLE tarifas_licencia (
    clase clase_licencia NOT NULL,
    vigencia_anos INT NOT NULL,
    costo NUMERIC(10, 2) NOT NULL,
    PRIMARY KEY (clase, vigencia_anos)
);

-- 7. Tabla de Configuración de Precios del Sistema
CREATE TABLE configuracion_sistema (
    clave VARCHAR(100) PRIMARY KEY,
    valor NUMERIC(10, 2) NOT NULL,
    descripcion VARCHAR(255)
);

-- 7. Índices de Optimización e Integridad de Reglas de Negocio

-- Garantiza que exista a lo sumo UNA sola licencia VIGENTE por titular y clase
CREATE UNIQUE INDEX idx_licencia_vigente_unica_por_clase 
ON licencias (titular_id, clase) 
WHERE (estado = 'VIGENTE');

-- Optimización de búsquedas requeridas por los listados y auditoría
CREATE INDEX idx_titulares_busqueda ON titulares (apellido, nombre);
CREATE INDEX idx_licencias_vencimiento ON licencias (fecha_vencimiento);
CREATE INDEX idx_titulares_filtros ON titulares (grupo_sanguineo, factor_rh, donante_organos);
